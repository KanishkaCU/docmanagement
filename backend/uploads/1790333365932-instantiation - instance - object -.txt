class car:

	mileage = 0
	no_of_airbags = 0


	def __init__ (self,no_of_wheels, no_of_airbags, mileage ##user input): ##constructor -- runs during every class call 

		print("This is a constructor")
		self.mileage = mileage
		self.no_of_wheels = no_of_wheels
		self.no_of_airbags = no_of_airbags
		

	def moveForward(self,speed): ## self  - object name 
		print("Car is moving with a speed of ",speed)

	def moveBackward(self):
		print("car is moving back")

	def __str__(self):
		return (self.carname) -- when we use string in default it prints hashcode so to print word we need to use it
